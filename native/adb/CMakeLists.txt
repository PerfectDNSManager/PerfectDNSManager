cmake_minimum_required(VERSION 3.22)
project(pdm_adb LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${BORINGSSL_SOURCE}" boringssl EXCLUDE_FROM_ALL)
add_library(adb SHARED adb_pairing.cpp)
target_include_directories(adb PRIVATE "${BORINGSSL_SOURCE}/include")
target_compile_options(adb PRIVATE -Os -fvisibility=hidden -ffunction-sections -fdata-sections -fstack-protector-strong)
target_link_options(adb PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,-z,relro -Wl,-z,now)
target_link_libraries(adb PRIVATE crypto log)
